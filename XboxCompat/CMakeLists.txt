cmake_minimum_required(VERSION 3.20)
project(XboxCompat LANGUAGES CXX)

# Static CRT: the shims must not depend on any redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(/W4 /permissive- /utf-8)

set(XC_OUT ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${XC_OUT})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${XC_OUT})
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${cfg} ${XC_OUT})
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${cfg} ${XC_OUT})
endforeach()

include_directories(common)

# xmem.dll -- Xbox title-memory API (XMemAlloc & friends) on top of the Windows heap.
add_library(xmem SHARED xmem/xmem.cpp xmem/xmem_phys.cpp xmem/xmem.def)
target_link_options(xmem PRIVATE /NOIMPLIB)
set_target_properties(xmem PROPERTIES PREFIX "" OUTPUT_NAME "xmem")

# d3d12_xs.dll -- Xbox Series D3D12 runtime name; forwards the evidenced public entry points to d3d12.dll.
add_library(d3d12_xs SHARED d3d12_xs/d3d12_xs.cpp)
target_link_options(d3d12_xs PRIVATE /NOIMPLIB)
set_target_properties(d3d12_xs PROPERTIES PREFIX "" OUTPUT_NAME "d3d12_xs")

# Pure traced-stub DLLs for Xbox-only modules imported by game_scarlett_master.exe (generated by tools/gen_stubs.py).
foreach(m xg_xs chad xapu AcpHal XCRDAPI)
  add_library(${m} SHARED generated/${m}_stubs.cpp)
  target_link_options(${m} PRIVATE /NOIMPLIB)
  set_target_properties(${m} PROPERTIES PREFIX "" OUTPUT_NAME "${m}")
endforeach()

# Self-test harness (loads only the shims built here).
add_executable(xc_selftest tests/test_shims.cpp)
add_dependencies(xc_selftest xmem d3d12_xs)
